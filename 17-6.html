function zxc_step(名称, 秒) {
  return new Promise(解决 => setTimeout(解决, 秒 * 1000)); // setTimeout 模拟耗时
}
async function zxc_run() {
  await zxc_step("买原料", 6);                 // 顺序执行
  await zxc_step("取原料", 2);
  await Promise.all([                          // 两项同时进行
    zxc_step("和面", 4),
    zxc_step("和馅", 3)
  ]);
  await zxc_step("包饺子", 3);
  await zxc_step("吃饺子", 2);
}
